\title{\bf Consistency of Optimal Matching-based Clustering\\
for Mixtures of Markov Chains}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
We study the consistency of clustering algorithms based on the Optimal Matching (edit) dissimilarity for categorical sequences drawn from a mixture of stationary ergodic processes on a finite alphabet.
We first establish the almost sure convergence of the normalised dissimilarity $\dom(X_{1:n},Y_{1:n})/n$ to a deterministic limit $\gamma(X,Y)$ via Kingman's subadditive ergodic theorem, and prove exponential concentration of this estimator around $\gamma$ for Markov chains using McDiarmid's inequality for dependent variables.
Under an explicit separation condition on the asymptotic pairwise dissimilarities, we show that single-linkage hierarchical clustering and $K$-medoids (PAM) both recover the true partition with probability tending to one as $n,N\to\infty$, and we make the exact recovery threshold $n=\Omega(\log N)$ explicit.
We provide computable upper and lower bounds on $\gamma$ via $1$-Wasserstein distances on the stationary distributions, and extend all results to a unified framework that covers Hidden Markov Models and multichannel observations as special cases.
All proofs are deferred to Section~\ref{sec:proofs}.
\end{abstract}

\section{The Optimal Matching Dissimilarity}
\label{sec:OM}

\subsection{Alphabet, Strings, and Edit Operations}


\begin{definition}[Alphabet and strings]
\label{def:alphabet}
Let $\Sigma$ be a finite alphabet with $|\Sigma|=d\ge 2$. For $n\ge 0$, write
$\Sigma^n$ for the set of strings $x_{1:n}=(x_1,\ldots,x_n)$ with letters in
$\Sigma$, with $\Sigma^0=\{\emptyset\}$ the empty string, and set
$\Sigma^\star:=\bigcup_{n\ge 0}\Sigma^n$. We write $\Sigma^{\N}$ for the set of
one-sided infinite sequences.
\end{definition}

\begin{definition}[Optimal Matching dissimilarity]
\label{def:OM}
Fix a symmetric substitution cost $c_{\sub}:\Sigma\times\Sigma\to[0,\infty)$ with $c_{\sub}(a,a)=0$, and insertion/deletion costs $c_{\ins},c_{\del}:\Sigma\to(0,\infty)$. For $x_{1:n},y_{1:m}\in\Sigma^\star$, an \emph{edit path} from $x_{1:n}$ to $y_{1:m}$ is a finite sequence of elementary operations---substitution of a letter $a$ by a letter $b$ (cost $c_{\sub}(a,b)$), deletion of a letter $a$ (cost $c_{\del}(a)$), insertion of a letter $b$ (cost $c_{\ins}(b)$)---applied successively and transforming $x_{1:n}$ into $y_{1:m}$. The cost of an edit path is the sum of the costs of its operations, and the \emph{OM dissimilarity} is
\[
\dom(x_{1:n},y_{1:m}) := \min\{\mathrm{cost}(\pi):\ \pi\text{ edit path from }x_{1:n}\text{ to }y_{1:m}\}.
\]
\end{definition}

\begin{assumption}[Metric cost scheme]
\label{ass:metric}
The costs $(c_{\ins},c_{\del},c_{\sub})$ satisfy:
\begin{enumerate}[label=(\roman*)]
\item $c_{\ins}(a)=c_{\del}(a)=:\delta(a)$ for all $a\in\Sigma$;
\item $c_{\sub}(a,b)=c_{\sub}(b,a)$ for all $a,b\in\Sigma$;
\item $c_{\sub}(a,c)\le c_{\sub}(a,b)+c_{\sub}(b,c)$ for all $a,b,c\in\Sigma$;
\item $c_{\sub}(a,b)\le\delta(a)+\delta(b)$ for all $a,b\in\Sigma$;
\item $\delta(a)\le c_{\sub}(a,b)+\delta(b)$ for all $a,b\in\Sigma$;
\item $c_{\sub}(a,b)>0$ for all $a\neq b$.
\end{enumerate}
\end{assumption}

Let $\Sigma_\bot:=\Sigma\cup\{\bot\}$, where $\bot$ denotes a gap symbol, and extend $c_{\sub}$ to $\Sigma_\bot$ by
\begin{equation}
\label{eq:dbar}
\bar d:\Sigma_\bot\times\Sigma_\bot\to\R^+,\qquad
\bar d(a,b):=\begin{cases} c_{\sub}(a,b) & a,b\in\Sigma,\\ \delta(a) & a\in\Sigma,\ b=\bot,\\ \delta(b) & a=\bot,\ b\in\Sigma,\\ 0 & a=b=\bot.\end{cases}
\end{equation}
Under Assumption~\ref{ass:metric}, $\bar d$ is a metric on $\Sigma_\bot$.

An \emph{alignment} of $(x_{1:n},y_{1:m})$ is a pair $(\tilde x_{1:L},\tilde y_{1:L})\in\Sigma_\bot^L\times\Sigma_\bot^L$ with $(\tilde x_\ell,\tilde y_\ell)\neq(\bot,\bot)$ for every $\ell$, such that deleting the $\bot$ symbols from $\tilde x$ recovers $x_{1:n}$ and deleting them from $\tilde y$ recovers $y_{1:m}$. Its cost is
\begin{equation}
\label{eq:aligncost}
\mathrm{cost}(\tilde x,\tilde y) := \sum_{\ell=1}^L \bar d(\tilde x_\ell,\tilde y_\ell).
\end{equation}
Under Assumption~\ref{ass:metric}, $\dom$ is a metric on $\Sigma^\star$ and coincides with the minimal alignment cost \cite{Bioinformatics}. We set
\begin{equation}
\label{eq:Mdef}
M := \max_{a,b\in\Sigma}c_{\sub}(a,b),
\end{equation}
so that $\dom(x_{1:n},y_{1:n})\le Mn$, by the in-place substitution alignment.
\subsection{Almost Sure Convergence}

Let $X=(X_t)_{t\ge 1}$ and $Y=(Y_t)_{t\ge 1}$ be independent processes on $\Sigma$. Set
\[
D_n := \dom(X_{1:n},Y_{1:n}).
\]

\begin{proposition}[Convergence of the normalised OM dissimilarity]
\label{prop:convergence}
Let $X$ and $Y$ be independent time-homogeneous Markov chains on $\Sigma$ with
irreducible kernels $P,Q$, stationary laws $\pi,\pi'$, and arbitrary initial
laws $\mu,\mu'$. Then $\hat\gamma_n:=D_n/n$ converges almost surely to a
deterministic limit $\gamma(P,Q)$, which does not depend on the initial laws
and is given by
\[
  \gamma(P,Q) \;=\; \inf_{n\ge 1}\frac{\E_{\pi,\pi'}[D_n]}{n}.
\]
\end{proposition}

\begin{remark}
\label{rmk:gamma-not-metric}
$\gamma$ is not a metric on laws : in general $\gamma(P,P) \ne 0$.
\end{remark}




\subsection{Concentration for Markov Chains}

The following follows from a McDiarmid-type inequality for Markov chains obtained by Marton couplings~\citep{paulin}.

\begin{proposition}[Exponential concentration]
\label{prop:concentration}
Let $X$ and $Y$ be independent time-homogeneous irreducible aperiodic Markov chains on $\Sigma$ with transition matrices $P$ and $Q$, and let $\tmix=\tmix(P\otimes Q)$ be the mixing time of the product chain on $\Sigma\times\Sigma$, in the sense of \cite{paulin}. There exists a universal constant $C_{\mathrm{Pau}}>0$ such that, setting
\begin{equation}
\label{eq:Cdef}
C=C(M,P,Q) := C_{\mathrm{Pau}}\,M^2\,\tmix,
\end{equation}
the estimator $\hat\gamma_n := D_n/n$ satisfies: for every $t>0$ there exists $n_0(t)$ such that, for every $n\ge n_0(t)$,
\[
\PP\bigl(|\hat\gamma_n-\gamma|\ge t\bigr) \;\le\; 2\exp\!\Bigl(-\frac{t^2 n}{2C}\Bigr).
\]
\end{proposition}


\subsection{A Regularity Property of $\gamma$}
\label{ssec:usc}

Since $\gamma$ governs the separation condition for clustering (Assumption~\ref{ass:separation} below), we record how it behaves under perturbations of the transition matrices. Let $\mathcal S^\circ_d$ denote the set of transition matrices of irreducible aperiodic Markov chains on $\Sigma$, endowed with the entrywise norm $\|P-P'\|_\infty=\max_{a,b}|P(a,b)-P'(a,b)|$.

\begin{proposition}[Upper semi-continuity of $\gamma$]
\label{prop:usc}
The map $\gamma:\mathcal S^\circ_d\times\mathcal S^\circ_d\to\R^+$ is upper semi-continuous: if $(P_m,Q_m)\to(P_0,Q_0)$, then
\[
\limsup_{m\to\infty}\gamma(P_m,Q_m) \;\le\; \gamma(P_0,Q_0).
\]
\end{proposition}

\begin{remark}
\label{rmk:usc-consequence}
For every $c>0$, the set $\{P\in\mathcal S^\circ_d:\gamma(P,P)<c\}$ is open: low within-cluster dispersion is stable under perturbation of the transition matrix. Whether $\gamma$ is continuous is left open.
\end{remark}

%% ======================================================================
\subsection{Bounds on $\gamma$}
\label{sec:bounds}
%% ======================================================================

\begin{proposition}[Wasserstein lower bound; product upper bound]
\label{prop:bounds}
Let $P,Q$ be irreducible transition matrices on $\Sigma$ with stationary
distributions $\pi_P,\pi_Q$, let $S:=(c_{\sub}(a,b))_{a,b\in\Sigma}$, and let
$\Wd$ denote the $1$-Wasserstein distance on $(\Sigma_\bot,\bar d)$. Then
\begin{equation}
\label{eq:bounds}
\Wd(\pi_P,\pi_Q) \;\le\; \gamma(P,Q) \;\le\; \pi_P^\top S\,\pi_Q .
\end{equation}
\end{proposition}

Both bounds depend on the chains through their stationary distributions alone,
and are computable from them: the upper bound in closed form, the lower one by
linear programming. The lower bound is informative in that
$\Wd(\pi_P,\pi_Q)>0$ as soon as $\pi_P\neq\pi_Q$, so $\gamma(P,Q)>0$ whenever
the two chains have different stationary distributions.

\section{Clustering Consistency}
\label{sec:clustering}

\subsection{The Mixture Model}

Let $\mathbb P_1,\ldots,\mathbb P_K$ be $K\ge 2$ time-homogeneous irreducible
aperiodic Markov chains on $\Sigma$, with arbitrary initial distributions and
mixture weights $w_1,\ldots,w_K\in(0,1)$, $\sum_k w_k=1$. We observe $N$
sequences $X_1,\ldots,X_N$ of length $n$, generated as follows : latent labels
$Z_1,\ldots,Z_N$ are drawn i.i.d.~with $\PP(Z_i=k)=w_k$, and conditionally on
$Z_i=k$, $X_i$ is a length-$n$ realisation of $\mathbb P_k$, independently
across $i$. Let $C_k:=\{i:Z_i=k\}$, $w_{\min}:=\min_k w_k$, $w_{\max}:=\max_k w_k$. Set
\[
\Gamma_{k,k'}:=\gamma(\mathbb P_k,\mathbb P_{k'}),
\qquad
\hat\gamma_n(i,j):=\frac{1}{n}\dom(X_i,X_j),
\]
the first quantity being well defined by Proposition~\ref{prop:convergence},
which also guarantees that it does not depend on the initial distributions. Let
$P^\star$ denote the partition of $\{1,\ldots,N\}$ into the \emph{non-empty}
classes $\{C_k:C_k\neq\emptyset\}$ induced by $(Z_i)_{i=1}^N$.

\begin{assumption}[Growth condition]
\label{ass:growth}
$N=N_n$ satisfies $N_n\to\infty$ and $\log N_n = o(n)$ as $n\to\infty$.
\end{assumption}

\begin{assumption}[Separation condition]
\label{ass:separation}
Let $\gin:=\max_k\Gamma_{kk}$ and $\gout:=\min_{k\neq k'}\Gamma_{kk'}$. We assume $\eta:=\gout-\gin>0$.
\end{assumption}

The interpretation of Assumption~\ref{ass:separation} is natural: the maximum within-cluster asymptotic dispersion $\gin$ must be strictly smaller than the minimum between-cluster asymptotic distance $\gout$.

The first key intermediate result is the uniform concentration of the empirical dissimilarity matrix around its limit.

\begin{lemma}[Uniform convergence of the dissimilarity matrix]
\label{lem:uniform}
Fix $\varepsilon>0$ and define the event
\[
\mathcal E_{N,n}(\varepsilon) := \Bigl\{\forall\, 1\le i<j\le N:\ |\hat\gamma_n(i,j)-\Gamma_{Z_i Z_j}|\le\varepsilon\Bigr\}.
\]
Set $C^\star := \max_{k,k'}C(M,\mathbb P_k,\mathbb P_{k'})$, with $C(\cdot)$ as in~\eqref{eq:Cdef}. Under Assumption~\ref{ass:metric}, for every $n\ge n_0(\varepsilon)$,
\[
\PP(\mathcal E_{N,n}(\varepsilon)^c) \;\le\; N^2\exp\!\Bigl(-\frac{\varepsilon^2 n}{2C^\star}\Bigr).
\]
In particular, under Assumption~\ref{ass:growth}, $\PP(\mathcal E_{N,n}(\varepsilon))\to 1$ as $n\to\infty$.
\end{lemma}


\subsection{Single-Linkage Hierarchical Clustering}

\begin{definition}[Hierarchical agglomerative clustering]
\label{def:HAC}
Let $L$ be a linkage, i.e.~a function assigning a non-negative number to every
pair of disjoint subsets of $\{1,\ldots,N\}$. Hierarchical agglomerative
clustering constructs a sequence of partitions $(\mathcal P_\ell)_{\ell=0}^{N-1}$
starting from $\mathcal P_0:=(\{1\},\ldots,\{N\})$ and, at step $\ell$, merging
a pair $(A_\ell,B_\ell)\in\arg\min_{C\neq D\in\mathcal P_\ell}L(C,D)$ to form
$\mathcal P_{\ell+1}=(\mathcal P_\ell\setminus\{A_\ell,B_\ell\})\cup\{A_\ell\cup B_\ell\}$.
The procedure terminates at $\mathcal P_{N-1}=\{\{1,\ldots,N\}\}$.
\end{definition}

For \emph{single linkage}, $L(A,B):=\min_{i\in A,j\in B}\hat\gamma_n(i,j)$.
Equivalently, for every $t\ge 0$ define the graph
$G_t:=(\{1,\ldots,N\},E_t)$ with $(i,j)\in E_t$ iff $\hat\gamma_n(i,j)\le t$;
the partition obtained by single linkage at level $t$ is the set of connected
components of $G_t$, denoted $\mathcal P^{\mathrm{SL}}_{N,n}(t)$.

\begin{theorem}[Consistency of single-linkage clustering]
\label{thm:SL}
Under Assumptions~\ref{ass:metric} and~\ref{ass:separation}, fix
$t\in(\gin,\gout)$ and $\varepsilon\in(0,\eta/2)$ small enough that
$\gin+\varepsilon<t<\gout-\varepsilon$. Then for every $n\ge n_0(\varepsilon)$
and every $N\ge 1$,
\[
\PP\bigl(\mathcal P^{\mathrm{SL}}_{N,n}(t)\neq P^\star\bigr)
\;\le\;N^2\exp\!\Bigl(-\frac{\varepsilon^2 n}{2C^\star}\Bigr).
\]
In particular, under Assumption~\ref{ass:growth},
$\PP\bigl(\mathcal P^{\mathrm{SL}}_{N,n}(t)=P^\star\bigr)\to 1$ as $n\to\infty$.
\end{theorem}

Any threshold lying strictly between the within- and between-cluster
asymptotic dissimilarities recovers the true partition exactly, with error
probability decaying exponentially in $n$. 

\subsection{Selecting the Number of Clusters}
\label{ssec:K-selection}

Theorem~\ref{thm:SL} assumes that $K$ is known. We show that under a mild
strengthening of Assumption~\ref{ass:separation}, the largest-gap rule applied
to the single-linkage dendrogram produces a consistent estimator of $K$.

Single linkage produces a sequence of partitions
$(\mathcal P_\ell)_{\ell=0}^{N-1}$ (Definition~\ref{def:HAC}) together with
$N-1$ \emph{merge heights} $h_1\le\cdots\le h_{N-1}$, where
$h_\ell:=\min_{C\neq D\in\mathcal P_{\ell-1}}L(C,D)$ is the linkage distance at
which the $\ell$-th merge occurs; after $\ell$ merges the partition
$\mathcal P_\ell$ has $N-\ell$ blocks. The \emph{largest-gap estimator} of $K$
is
\begin{equation}
\label{eq:Khat-gap}
\hat K_{N,n} \;:=\; N - \ell^\star,
\qquad
\ell^\star := \min\Bigl(\arg\max_{1\le \ell\le N-2}(h_{\ell+1}-h_\ell)\Bigr),
\end{equation}
that is, the number of blocks remaining just before the largest jump in the
merge-height sequence, taking the smallest index in case of tie. The
associated partition $\hat{\mathcal P}^{\mathrm{SL}}_{N,n}:=\mathcal P_{\ell^\star}$
is the corresponding cut of the dendrogram. By construction
$\hat K_{N,n}\in\{2,\ldots,N-1\}$, so the estimator implicitly assumes
$K\ge 2$.


\begin{assumption}[Strong separation]
\label{ass:strong-separation}
Let $\gout^{\max}:=\max_{k\neq k'}\Gamma_{kk'}$. We assume
\begin{align}
\label{eq:strong-sep-1}
\gout - \gin &> \gin, \\
\label{eq:strong-sep-2}
\gout - \gin &> \gout^{\max} - \gout.
\end{align}
\end{assumption}

This is a strengthening of Assumption~\ref{ass:separation}, which only required $\gout - \gin > 0$. Condition~\eqref{eq:strong-sep-1} controls the within-cluster dispersion and condition~\eqref{eq:strong-sep-2} controls the dispersion of the between-cluster distances. Assumption~\ref{ass:strong-separation} only intervenes in the present subsection and is not required by any other result in the paper.

\begin{theorem}[Consistency of the largest-gap rule]
\label{thm:K-selection}
Under Assumptions~\ref{ass:metric}--\ref{ass:separation} and~\ref{ass:strong-separation}, and with $K\ge2$,
\[
\PP\bigl(\hat K_{N,n}=K\bigr) \xrightarrow[n\to\infty]{} 1,
\qquad
\PP\bigl(\hat{\mathcal P}^{\mathrm{SL}}_{N,n}=P^\star\bigr) \xrightarrow[n\to\infty]{} 1.
\]
More precisely, for $\varepsilon$ as in~\eqref{eq:eps-Kselect} and $n\ge n_0(\varepsilon)$, the probability that either conclusion fails is at most $N^2\exp(-\varepsilon^2n/(2C^\star))+K(1-w_{\min})^N$.
\end{theorem}


\subsection{$K$-Medoids (PAM)}

\begin{definition}[$K$-medoids]
\label{def:kmedoids}
The $K$-medoids estimator is a minimiser
\[
\hat M_{N,n} \in \arg\min_{\substack{M\subseteq\{1,\ldots,N\}\\|M|=K}}\Phi_{N,n}(M),\qquad
\Phi_{N,n}(M) := \frac{1}{N}\sum_{i=1}^N \min_{m\in M}\hat\gamma_n(i,m).
\]
The associated partition $\hat P_{N,n}$ assigns each $i\in\{1,\ldots,N\}$ to its closest medoid, taking a random medoid in case of tie.
\end{definition}
\begin{assumption}[Balance condition]
\label{ass:balance}
$\gout - \gin > \gin\,\dfrac{w_{\max}}{w_{\min}}$. Equivalently,
\begin{equation}
\label{eq:Adef}
A \;:=\; w_{\min}(\gout-\gin)-w_{\max}\gin
\end{equation}
is strictly positive.
\end{assumption}

Assumption~\ref{ass:balance} is stronger than
Assumption~\ref{ass:separation}: the separation gap must dominate the
within-cluster dispersion by a factor controlled by the weight imbalance. When
the mixture is balanced ($w_k=1/K$), it reduces to $\gout>2\gin$.

In the sequel, fix $\varepsilon>0$ and $\delta_0>0$ satisfying
\begin{equation}
\label{eq:eps-kmedoids}
0 < \varepsilon < \min\!\left(\frac{\eta}{2},\;\frac{A}{2(3+w_{\max})}\right),
\qquad
\delta_0 := \min\!\left(\frac{A}{4M},\ \frac{w_{\min}}{2}\right),
\end{equation}
both positive under Assumption~\ref{ass:balance}, and define the
\emph{label-frequency event}
\begin{equation}
\label{eq:Wevent}
\mathcal W_N(\delta_0) := \Bigl\{\forall\ell\in\{1,\ldots,K\}:\ |\hat w_\ell-w_\ell|\le\delta_0\Bigr\},
\qquad \hat w_\ell:=\frac{|C_\ell|}{N}.
\end{equation}
Since the $Z_i$ are i.i.d., Hoeffding's inequality and a union bound give
\begin{equation}
\label{eq:Wbound}
\PP\bigl(\mathcal W_N(\delta_0)^c\bigr)\;\le\;2K\exp\bigl(-2N\delta_0^2\bigr).
\end{equation}
On $\mathcal W_N(\delta_0)$ one has $\hat w_\ell\ge w_{\min}-\delta_0\ge w_{\min}/2>0$
for every $\ell$; in particular no class is empty.

\begin{lemma}[Every cluster has exactly one medoid]
\label{lem:kmedoid-one-per-cluster}
Under Assumptions~\ref{ass:metric}--\ref{ass:balance}, on the event
$\mathcal E_{N,n}(\varepsilon)\cap\mathcal W_N(\delta_0)$ with
$\varepsilon,\delta_0$ as in~\eqref{eq:eps-kmedoids}, every minimiser $M$ of
$\Phi_{N,n}$ contains exactly one index from each cluster $C_1,\ldots,C_K$.
\end{lemma}

\begin{theorem}[Consistency of $K$-medoids]
\label{thm:kmedoids}
Under Assumptions~\ref{ass:metric}--\ref{ass:balance}, with $\varepsilon,\delta_0$ as in~\eqref{eq:eps-kmedoids}, for every $n\ge n_0(\varepsilon)$,
\[
\PP\bigl(\hat P_{N,n}\neq P^\star\bigr)\;\le\;N^2\exp\!\Bigl(-\frac{\varepsilon^2n}{2C^\star}\Bigr)\;+\;2K\exp\bigl(-2N\delta_0^2\bigr).
\]
In particular, under Assumption~\ref{ass:growth}, $\PP(\hat P_{N,n}=P^\star)\to1$ as $n,N\to\infty$.
\end{theorem}


\begin{remark}[Estimating $K$ beforehand]
\label{rmk:K-then-kmedoids}
Theorems~\ref{thm:K-selection} and~\ref{thm:kmedoids} combine into a procedure
requiring no prior knowledge of $K$: estimate $\hat K_{N,n}$ from the
dendrogram via~\eqref{eq:Khat-gap}, then run $K$-medoids with $\hat K_{N,n}$
medoids. On $\{\hat K_{N,n}=K\}$ the second step is exactly
Definition~\ref{def:kmedoids}, so under
Assumptions~\ref{ass:metric}--\ref{ass:balance}
and~\ref{ass:strong-separation} the probability that the resulting partition
differs from $P^\star$ is at most the sum of the bounds of the two theorems,
and tends to $0$.
\end{remark}

\subsection{Exact Recovery Threshold}

\begin{corollary}[Exact recovery]
\label{cor:recovery}
Let Assumptions~\ref{ass:metric} and~\ref{ass:separation} hold, fix
$\varepsilon\in(0,\eta/2)$ and $\theta>0$, and suppose
\begin{equation}
\label{eq:recovery-n}
n\;\ge\;\max\Bigl(n_0(\varepsilon),\ \frac{4C^\star(1+\theta)}{\varepsilon^2}\,\log N\Bigr).
\end{equation}
Then for every $t$ with $\gin+\varepsilon<t<\gout-\varepsilon$,
\[
\PP\bigl(\mathcal P^{\mathrm{SL}}_{N,n}(t)\neq P^\star\bigr)\;\le\;N^{-2\theta},
\]
and, under Assumption~\ref{ass:balance} with $\varepsilon,\delta_0$ as
in~\eqref{eq:eps-kmedoids},
\[
\PP\bigl(\hat P_{N,n}\neq P^\star\bigr)\;\le\;N^{-2\theta}+2K\exp\bigl(-2N\delta_0^2\bigr).
\]
\end{corollary}

Exact recovery therefore holds as soon as $n=\Omega(\log N)$, with a
multiplicative constant $4C^\star(1+\theta)/\varepsilon^2$ reflecting the union
bound over the $\binom{N}{2}$ pairwise dissimilarities.

\section{An Extension to Multichannel Hidden Markov Models}
\label{sec:extensions}

The clustering results of Section~\ref{sec:clustering} rely on two
ingredients: (i) the exponential concentration of $\hat\gamma_n(i,j)$ around
$\Gamma_{Z_iZ_j}$ (Proposition~\ref{prop:concentration}), and (ii) a
deterministic combinatorial argument conditional on
$\mathcal E_{N,n}(\varepsilon)$ and $\mathcal W_N(\delta_0)$. Ingredient~(ii)
uses no property of the observed sequences beyond those two events, together
with Hoeffding's inequality for the i.i.d.\ labels. Ingredient~(i) requires
only that the observed sequences be measurable functions of an underlying
Markov chain with finite mixing time, a setting explicitly covered
by~\citet[Corollary~2.17]{paulin}. We now phrase this observation as a single
theorem covering Hidden Markov Models and multichannel observations.

\begin{definition}[Generalised observation model]
\label{def:generalised}
Each component $k\in\{1,\ldots,K\}$ is specified by:
\begin{enumerate}[label=(\roman*)]
\item a finite hidden state space $\mathcal Z$ and an irreducible aperiodic
Markov chain $(Z^{(k)}_t)_{t\ge 1}$ on $\mathcal Z$, with arbitrary initial
law;
\item a possibly multichannel observation alphabet
$\Sigma=\Sigma_1\times\cdots\times\Sigma_C$ with $C\ge 1$;
\item a measurable emission map $\varphi_k:\mathcal Z\times\mathcal U\to\Sigma$,
where $(\mathcal U,\nu)$ is a Polish noise space; the observed process is
$X^{(k)}_t:=\varphi_k(Z^{(k)}_t,\xi_t)$ with
$(\xi_t)\stackrel{\text{iid}}{\sim}\nu$ independent of the hidden chain.
\end{enumerate}
The OM dissimilarity is computed on $\Sigma$ with the multichannel substitution
cost $c^{\mathrm{mc}}_{\sub}(a,b):=\sum_{c=1}^C \lambda_c\,c^{(c)}_{\sub}(a_c,b_c)$
and multichannel gap cost
$\delta^{\mathrm{mc}}(a):=\sum_{c=1}^C \lambda_c\,\delta^{(c)}(a_c)$, where
$\lambda_1,\ldots,\lambda_C>0$ are channel weights. Set
$M^{\mathrm{mc}}:=\max_{a,b}c^{\mathrm{mc}}_{\sub}(a,b)$.
\end{definition}

Assumption~\ref{ass:metric} is stable under this channelwise aggregation: if
each $(c^{(c)}_{\sub},\delta^{(c)})$ satisfies (i)--(vi) on $\Sigma_c$, then so
does $(c^{\mathrm{mc}}_{\sub},\delta^{\mathrm{mc}})$ on $\Sigma$, by taking
positive combinations of the corresponding inequalities.

\begin{proposition}[Convergence and concentration under the generalised model]
\label{prop:gen-concentration}
Assume Definition~\ref{def:generalised} and Assumption~\ref{ass:metric} applied
to $(c^{\mathrm{mc}}_{\sub},\delta^{\mathrm{mc}})$ on $\Sigma$, and let
$X^{(k)}$ and $X^{(k',\prime)}$ be independent realisations of components $k$
and $k'$. Then:
\begin{enumerate}[label=(\roman*),leftmargin=*,topsep=2pt]
\item the limit
$\gamma^{\mathrm{obs}}(k,k'):=\lim_{n}\tfrac1n\dom(X^{(k)}_{1:n},X^{(k',\prime)}_{1:n})$
exists almost surely, is deterministic, and depends only on the laws of the
observed processes;
\item with $\hat\gamma^{\mathrm{obs}}_n:=\tfrac1n\dom(X^{(k)}_{1:n},X^{(k',\prime)}_{1:n})$
and $C^{\mathrm{gen}}:=C_{\mathrm{Pau}}(M^{\mathrm{mc}})^2\,\tmix^{\mathrm{hid}}$,
where $\tmix^{\mathrm{hid}}$ is the mixing time of the product hidden chain
$(Z^{(k)},Z^{(k',\prime)})$ on $\mathcal Z\times\mathcal Z$, for every $t>0$
there exists $n_0(t)$ such that, for every $n\ge n_0(t)$,
\[
\PP\bigl(|\hat\gamma^{\mathrm{obs}}_n-\gamma^{\mathrm{obs}}(k,k')|\ge t\bigr)
\;\le\; 2\exp\!\Bigl(-\frac{t^2 n}{2 C^{\mathrm{gen}}}\Bigr).
\]
\end{enumerate}
\end{proposition}

\begin{theorem}[Clustering consistency under the generalised model]
\label{thm:gen-clustering}
Under Definition~\ref{def:generalised}, Assumption~\ref{ass:metric} applied to
$(c^{\mathrm{mc}}_{\sub},\delta^{\mathrm{mc}})$, and the analogues of
Assumptions~\ref{ass:growth}--\ref{ass:separation} formulated in terms of
$\gamma^{\mathrm{obs}}$:
\begin{enumerate}[label=(\roman*),leftmargin=*,topsep=2pt]
\item (Single linkage) for every
$t\in(\gin^{\mathrm{obs}},\gout^{\mathrm{obs}})$,
$\PP(\mathcal P^{\mathrm{SL}}_{N,n}(t)=P^\star)\to 1$;
\item ($K$-medoids) under the analogue of Assumption~\ref{ass:balance},
$\PP(\hat P_{N,n}=P^\star)\to 1$.
\end{enumerate}
The exact recovery threshold of Corollary~\ref{cor:recovery} holds with
$C^\star$ replaced by $C^{\mathrm{gen}}$.
\end{theorem}

\begin{remark}
Distinct specifications may induce the same law on the observed process
\cite{gilbert1959}; then
$\gamma^{\mathrm{obs}}(k,k')=\gamma^{\mathrm{obs}}(k,k)$ and the separation
condition fails. This is inherent: the OM dissimilarity operates on observed
sequences and cannot distinguish observationally equivalent models. The
separation condition is therefore a condition on the laws of the observed
processes, which is the natural clustering criterion.
\end{remark}

\subsection{Proofs for Section~\ref{sec:OM}}


\begin{proof}[Proof of Proposition~\ref{prop:convergence}]
\emph{Step 1: the stationary case.} We first prove the convergence for two
independent stationary ergodic processes $X,Y$ on $\Sigma$, which covers the
stationary case $(\mu,\mu')=(\pi,\pi')$. We may assume the processes are two-sided,
replacing each by its natural extension. Write $\mu_X:=\mathcal L(X)$,
$\nu_Y:=\mathcal L(Y)$, let $S$ denote the shift on $\Sigma^\Z$ and
$T:=S\times S$ the joint shift, which preserves $\mu_X\otimes\nu_Y$.

Concatenating an optimal alignment of $(X_{1:n},Y_{1:n})$ with an optimal
alignment of $(X_{n+1:n+m},Y_{n+1:n+m})$ yields an alignment of
$(X_{1:n+m},Y_{1:n+m})$ of cost $D_n+D_m\circ T^n$, hence
\begin{equation}
\label{eq:subadd}
D_{n+m} \;\le\; D_n + D_m\circ T^n .
\end{equation}
Moreover $0\le D_n\le Mn$. Kingman's subadditive ergodic
theorem~\citep{kingman} yields the almost sure and $L^1$ convergence of
$\hat\gamma_n$ to a limit $g$ with $\E[g]=\inf_{n\ge1}\E[D_n]/n$.

It remains to see that $g$ is deterministic. Set
$\Delta:=\max_{a\in\Sigma}\delta(a)$ and fix $k\ge1$. Deleting
$y_1,\ldots,y_k$ and then inserting $y_{n+1},\ldots,y_{n+k}$ transforms
$y_{1:n}$ into $y_{k+1:k+n}$, so $\dom(y_{1:n},y_{k+1:k+n})\le 2k\Delta$; the
same construction applied to the first argument gives
$\dom(x_{1:n},x_{k+1:k+n})\le 2k\Delta$. By the triangle inequality for
$\dom$, for all $x,y$ and all $n$,
\[
  \bigl|D_n(x,y)-D_n(x,S^ky)\bigr| \le 2k\Delta,
  \qquad
  \bigl|D_n(x,y)-D_n(S^kx,y)\bigr| \le 2k\Delta .
\]
Dividing by $n$, the limit $g$ is invariant under $\mathrm{id}\times S$ and
under $S\times\mathrm{id}$. By Fubini and ergodicity of $\mu_X$, for
$\nu_Y$-almost every $y$ the map $x\mapsto g(x,y)$ is $\mu_X$-almost surely
constant, equal to $h(y):=\int g(x,y)\,\mu_X(dx)$; the invariance under
$\mathrm{id}\times S$ then gives $h\circ S=h$ $\nu_Y$-almost everywhere, so $h$
is constant by ergodicity of $\nu_Y$. Thus $g=\E[g]$ almost surely.

Applied to two independent stationary chains with kernels $P$ and $Q$, which
are ergodic under the shift by irreducibility, this gives
$\hat\gamma_n\to\gamma(P,Q)=\inf_{n\ge1}\E_{\pi,\pi'}[D_n]/n$,
$\PP_{\pi,\pi'}$-almost surely.

\emph{Step 2: change of initial law.} We show that the convergence obtained in
Step~1 for $(\mu,\mu')=(\pi,\pi')$ transfers to arbitrary initial laws. Since
$\Sigma$ is finite and $P$ is irreducible, $\pi$ has full support, so
$\rho:\Sigma^\N\to\R^+$, $\rho(x):=\mu(x_1)/\pi(x_1)$, is well defined and
bounded.

For $n\ge1$ and $a_{1:n}\in\Sigma^n$, let
$C(a_{1:n}):=\{x\in\Sigma^\N: x_{1:n}=a_{1:n}\}$ denote the corresponding
cylinder. By the Markov property and time-homogeneity of the kernel,
\[
  \PP_\mu\bigl(C(a_{1:n})\bigr) = \mu(a_1)\prod_{t=1}^{n-1}P(a_t,a_{t+1}),
  \qquad
  \PP_\pi\bigl(C(a_{1:n})\bigr) = \pi(a_1)\prod_{t=1}^{n-1}P(a_t,a_{t+1}),
\]
On $C(a_{1:n})$ the function $\rho$ is constant, equal to
$\mu(a_1)/\pi(a_1)$, so
\begin{equation}
\label{eq:cyl}
  \E_\pi\bigl[\rho\,\mathbf 1_{C(a_{1:n})}\bigr]
  = \frac{\mu(a_1)}{\pi(a_1)}\,\PP_\pi\bigl(C(a_{1:n})\bigr)
  = \mu(a_1)\prod_{t=1}^{n-1}P(a_t,a_{t+1})
  = \PP_\mu\bigl(C(a_{1:n})\bigr).
\end{equation}

The weight $\rho$ is the same
for all $n$: it is $\sigma(X_1)$-measurable, hence independent of the horizon.

Let now
\[
  \mathcal D := \bigl\{ A\in\mathcal F : \PP_\mu(A)=\E_\pi[\rho\,\mathbf 1_A]\bigr\},
\]
where $\mathcal F$ is the product $\sigma$-algebra on $\Sigma^\N$. The
cylinders, together with $\Sigma^\N$ itself, form a $\pi$-system generating
$\mathcal F$. By
\eqref{eq:cyl} this $\pi$-system is contained in $\mathcal D$. Moreover,
$\mathcal D$ is a $\lambda$-system. Dynkin's lemma therefore gives $\mathcal D=\mathcal F$, that is,
$\PP_\mu\ll\PP_\pi$ on the whole of $\mathcal F$, with
$d\PP_\mu/d\PP_\pi=\rho$.

Since $X$ and $Y$ are independent under both $\PP_{\mu,\mu'}$ and
$\PP_{\pi,\pi'}$, the same holds for the product laws:
$\PP_{\mu,\mu'}\ll\PP_{\pi,\pi'}$ on the product $\sigma$-algebra of
$\Sigma^\N\times\Sigma^\N$, with density $\rho\otimes\rho'$.

Each $D_n$ is a measurable function of $(X_{1:n},Y_{1:n})$, so
$A:=\{\hat\gamma_n\to\gamma(P,Q)\}$ is measurable for that $\sigma$-algebra. By
Step~1, $\PP_{\pi,\pi'}(A^c)=0$, so $(\rho\otimes\rho')\mathbf 1_{A^c}$
vanishes $\PP_{\pi,\pi'}$-almost surely and
\[
  \PP_{\mu,\mu'}(A^c)
  = \E_{\pi,\pi'}\bigl[(\rho\otimes\rho')\,\mathbf 1_{A^c}\bigr]
  = 0 .
\]
Hence $\hat\gamma_n\to\gamma(P,Q)$ $\PP_{\mu,\mu'}$-almost surely, with the
same limit as in the stationary case.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:concentration}]
The pair $Z=(X,Y)$ is an irreducible aperiodic Markov chain on $\Sigma^2$. In particular Proposition~\ref{prop:convergence} applies and $\gamma$ is deterministic. Define $f:(\Sigma^2)^n\to\R$ by
\[
f\bigl((x_1,y_1),\ldots,(x_n,y_n)\bigr) := \frac{1}{n}\dom(x_{1:n},y_{1:n}).
\]

We show that $f$ has bounded differences $c_i=2M/n$ in each coordinate, i.e., for every $z,z'\in(\Sigma^2)^n$,
\begin{equation}
\label{eq:bd}
|f(z)-f(z')| \;\le\; \sum_{i=1}^n c_i\,\mathbf 1_{z_i\neq z'_i},\qquad c_i=\frac{2M}{n}.
\end{equation}
Write $z=((x_t,y_t))_{t\le n}$ and $z'=((x'_t,y'_t))_{t\le n}$. The triangle inequality for $\dom$ gives
\[
\bigl|\dom(x,y)-\dom(x',y')\bigr| \;\le\; \dom(x,x') + \dom(y,y').
\]
The in-place substitution alignment yields $\dom(x,x')\le\sum_{t:x_t\neq x'_t}c_{\sub}(x_t,x'_t)\le M\,\dH(x,x')$, where $\dH$ is the Hamming distance, and similarly for $\dom(y,y')$. Hence
\[
n\,|f(z)-f(z')| \;\le\; M\bigl(\dH(x,x')+\dH(y,y')\bigr) \;\le\; 2M\,\dH(z,z'),
\]
which is~\eqref{eq:bd}.

Paulin's concentration inequality obtained by Marton coupling~\citep{paulin} provides a universal constant $C_{\mathrm{Pau}}>0$ such that, for every function $g$ on $(\Sigma^2)^n$ with bounded differences $(c_i)_{i\le n}$ in the sense of~\eqref{eq:bd} and every $s\ge 0$,
\begin{equation}
\label{eq:paulin-generic}
\PP\bigl(|g(Z_{1:n})-\E[g(Z_{1:n})]|\ge s\bigr) \;\le\; 2\exp\!\Bigl(-\frac{8s^2}{C_{\mathrm{Pau}}\,\tmix\sum_{i=1}^n c_i^2}\Bigr).
\end{equation}
Applying this to $f$ with $\sum_{i=1}^n c_i^2 = n\cdot(2M/n)^2 = 4M^2/n$,
\begin{equation}
\label{eq:concentration-centered}
\PP\bigl(|\hat\gamma_n-\E\hat\gamma_n|\ge s\bigr) \;\le\; 2\exp\!\Bigl(-\frac{2s^2 n}{C_{\mathrm{Pau}}M^2\tmix}\Bigr).
\end{equation}

Proposition~\ref{prop:convergence} gives $\hat\gamma_n\to\gamma$ almost surely. Since $0\le\hat\gamma_n\le M$, dominated convergence yields $\E\hat\gamma_n\to\gamma$. Hence, for every $t>0$ there exists $n_0=n_0(t)$ such that $|\E\hat\gamma_n-\gamma|\le t/2$ for all $n\ge n_0$. For such $n$, the triangle inequality $|\hat\gamma_n-\gamma|\le|\hat\gamma_n-\E\hat\gamma_n|+|\E\hat\gamma_n-\gamma|$ implies $\{|\hat\gamma_n-\gamma|\ge t\}\subseteq\{|\hat\gamma_n-\E\hat\gamma_n|\ge t/2\}$. Combining with~\eqref{eq:concentration-centered} applied to $s=t/2$,
\[
\PP\bigl(|\hat\gamma_n-\gamma|\ge t\bigr) \;\le\; 2\exp\!\Bigl(-\frac{t^2 n}{2\,C_{\mathrm{Pau}}M^2\tmix}\Bigr) \;=\; 2\exp\!\Bigl(-\frac{t^2 n}{2C}\Bigr),
\]
with $C$ as in~\eqref{eq:Cdef}.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{prop:usc}]
For $n\ge1$, set
\[
g_n(P,Q) := \frac{1}{n}\,\E_{P,Q}\bigl[\dom(X_{1:n},Y_{1:n})\bigr],
\]
the expectation being taken under the law of two independent stationary Markov chains with transitions $P$ and $Q$, so that $\gamma=\inf_{n\ge1}g_n$ by Proposition~\ref{prop:convergence}.

Each $g_n$ is continuous on $\mathcal S^\circ_d\times\mathcal S^\circ_d$. Indeed, on $\mathcal S^\circ_d$ the linear system $\pi_P P=\pi_P$, $\sum_a\pi_P(a)=1$ uniquely determines $\pi_P$, and Cramer's rule expresses $\pi_P$ as a rational function of the entries of $P$ with non-vanishing denominator on $\mathcal S^\circ_d$. Hence $P\mapsto\pi_P$ is continuous, and for each fixed $x_{1:n}\in\Sigma^n$,
\[
\mu_P^{(n)}(x_{1:n}) = \pi_P(x_1)\prod_{t=1}^{n-1}P(x_t,x_{t+1})
\]
is continuous in $P$. Since $\Sigma^n$ is finite,
\[
g_n(P,Q) = \frac{1}{n}\sum_{x,y\in\Sigma^n}\dom(x,y)\,\mu_P^{(n)}(x)\,\mu_Q^{(n)}(y)
\]
is a finite sum of continuous functions of $(P,Q)$, hence continuous.

Thus, $\gamma = \inf_{n\ge1} g_n$ is the pointwise infimum of a family of continuous functions, and any such infimum is upper semi-continuous. Explicitly: for $(P_m,Q_m)\to(P_0,Q_0)$ and any $\varepsilon>0$, choose $n_0$ such that $g_{n_0}(P_0,Q_0)<\gamma(P_0,Q_0)+\varepsilon$. By continuity of $g_{n_0}$, $g_{n_0}(P_m,Q_m)\to g_{n_0}(P_0,Q_0)$, and since $\gamma(P_m,Q_m)\le g_{n_0}(P_m,Q_m)$,
\[
\limsup_{m\to\infty}\gamma(P_m,Q_m)\le g_{n_0}(P_0,Q_0)<\gamma(P_0,Q_0)+\varepsilon.
\]
Letting $\varepsilon\to0$ concludes.
\end{proof}



\begin{proof}[Proof of Proposition~\ref{prop:bounds}]

By Proposition~\ref{prop:convergence}, $\gamma(P,Q)$ does not depend on the
initial laws; we may therefore take $X$ and $Y$ stationary throughout.

\emph{Lower bound.} $(\Sigma_\bot,\bar d)$ is a metric space. Let $f:\Sigma_\bot\to\R$ be $1$-Lipschitz with respect to $\bar d$. Fix $n\ge 1$ and an alignment $(\tilde x_{1:L},\tilde y_{1:L})\in\Sigma_\bot^L\times\Sigma_\bot^L$ of $(X_{1:n},Y_{1:n})$. By the $1$-Lipschitz property and the triangle inequality,
\[
\mathrm{cost}(\tilde x,\tilde y) = \sum_{\ell=1}^L \bar d(\tilde x_\ell,\tilde y_\ell) \;\ge\; \sum_{\ell=1}^L |f(\tilde x_\ell)-f(\tilde y_\ell)| \;\ge\; \Bigl|\sum_{\ell=1}^L f(\tilde x_\ell) - \sum_{\ell=1}^L f(\tilde y_\ell)\Bigr|.
\]
The aligned sequence $\tilde x$ contains the $n$ original symbols of $X_{1:n}$ together with $L-n$ copies of $\bot$; the same holds for $\tilde y$. The $f(\bot)$ terms cancel between the two sums, and we obtain
\[
\mathrm{cost}(\tilde x,\tilde y) \;\ge\; \Bigl|\sum_{t=1}^n f(X_t) - \sum_{t=1}^n f(Y_t)\Bigr|.
\]
The right-hand side does not depend on the alignment, so taking the minimum over alignments,
\[
\dom(X_{1:n},Y_{1:n}) \;\ge\; \Bigl|\sum_{t=1}^n f(X_t) - \sum_{t=1}^n f(Y_t)\Bigr|.
\]
Taking expectations, dividing by $n$, and applying Jensen's inequality to bring the expectation inside the absolute value, then using stationarity of $X,Y$:
\[
\frac{\E[D_n]}{n} \;\ge\; \E\Bigl|\tfrac{1}{n}\sum_{t=1}^n f(X_t) - \tfrac{1}{n}\sum_{t=1}^n f(Y_t)\Bigr| \;\ge\; \bigl|\E[f(X_1)]-\E[f(Y_1)]\bigr|.
\]
Letting $n\to\infty$, $\gamma(P,Q)\ge|\E_{\pi_P}[f]-\E_{\pi_Q}[f]|$. Taking the supremum over all $1$-Lipschitz $f:\Sigma_\bot\to\R$ and applying Kantorovich--Rubinstein duality on the finite metric space $(\Sigma_\bot,\bar d)$ yields $\gamma(P,Q)\ge\Wd(\pi_P,\pi_Q)$.

\emph{Upper bound.} The in-place substitution alignment of $(X_{1:n},Y_{1:n})$ has length $L=n$ (no gaps) and cost $\sum_{t=1}^n c_{\sub}(X_t,Y_t)$. Hence $\dom(X_{1:n},Y_{1:n})\le\sum_{t=1}^n c_{\sub}(X_t,Y_t)$. Dividing by $n$, taking expectations, and using stationarity and the independence of $X$ and $Y$:
\[
\frac{\E[D_n]}{n} \;\le\; \E[c_{\sub}(X_1,Y_1)] \;=\; \sum_{a,b\in\Sigma}\pi_P(a)\pi_Q(b)c_{\sub}(a,b) \;=\; \pi_P^\top S\,\pi_Q.
\]
Letting $n\to\infty$ gives the claim.
\end{proof}




\subsection{Proofs for Section~\ref{sec:clustering}}

\begin{proof}[Proof of Lemma~\ref{lem:uniform}]
For each pair $(i,j)$ with $i<j$, conditionally on $(Z_i,Z_j)=(k,k')$, the sequences $X_i$ and $X_j$ are independent realisations of $\mathbb P_k$ and $\mathbb P_{k'}$. Proposition~\ref{prop:concentration} gives, for $n\ge n_0(\varepsilon)$,
\[
\PP\bigl(|\hat\gamma_n(i,j)-\Gamma_{kk'}|>\varepsilon\,\big|\, Z_i=k,Z_j=k'\bigr) \;\le\; 2\exp\!\bigl(-\varepsilon^2 n/(2C(M,\mathbb P_k,\mathbb P_{k'}))\bigr),
\]
where $C(M,\mathbb P_k,\mathbb P_{k'})$ is the constant~\eqref{eq:Cdef} associated with the product chain $\mathbb P_k\otimes\mathbb P_{k'}$. Using $C(M,\mathbb P_k,\mathbb P_{k'})\le C^\star$ uniformly over the finitely many pairs $(k,k')$, and taking $n_0(\varepsilon)$ to be the maximum of the finitely many thresholds,
\[
\PP\bigl(|\hat\gamma_n(i,j)-\Gamma_{kk'}|>\varepsilon\,\big|\, Z_i=k,Z_j=k'\bigr) \;\le\; 2\exp\!\bigl(-\varepsilon^2 n/(2C^\star)\bigr).
\]
Marginalising over $(Z_i,Z_j)$ and applying a union bound over the $\binom{N}{2}\le N^2/2$ pairs:
\[
\PP(\mathcal E_{N,n}(\varepsilon)^c) \;\le\; \binom{N}{2}\cdot 2\exp\!\bigl(-\varepsilon^2 n/(2C^\star)\bigr) \;\le\; N^2\exp\!\bigl(-\varepsilon^2 n/(2C^\star)\bigr).
\]
Under Assumption~\ref{ass:growth}, $\log(N^2)=2\log N=o(n)$, so the right-hand side tends to $0$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:SL}]
Work on the event $\mathcal E_{N,n}(\varepsilon)$. On this event:
\begin{itemize}[leftmargin=*,topsep=2pt]
\item If $Z_i\neq Z_j$, then $\hat\gamma_n(i,j)\ge\Gamma_{Z_i Z_j}-\varepsilon\ge\gout-\varepsilon>t$, so $(i,j)\notin E_t$. Therefore each connected component of $G_t$ is contained in a single class $C_k$.
\item If $Z_i=Z_j=k$, then $\hat\gamma_n(i,j)\le\Gamma_{kk}+\varepsilon\le\gin+\varepsilon<t$, so $(i,j)\in E_t$. Hence each non-empty $C_k$ is contained in a single connected component (any two of its vertices are joined by an edge).
\end{itemize}
Combining, the connected components of $G_t$ are exactly the non-empty classes $C_k$, that is, $\mathcal P^{\mathrm{SL}}_{N,n}(t)=P^\star$. Therefore
\[
\PP\bigl(\mathcal P^{\mathrm{SL}}_{N,n}(t)\neq P^\star\bigr) \;\le\; \PP(\mathcal E_{N,n}(\varepsilon)^c)\;\le\;N^2\exp\!\Bigl(-\frac{\varepsilon^2n}{2C^\star}\Bigr)
\]
by Lemma~\ref{lem:uniform}, which tends to $0$ under Assumption~\ref{ass:growth}.
\end{proof}



\begin{proof}[Proof of Theorem~\ref{thm:K-selection}]
Pick $\varepsilon>0$ small enough that
\begin{equation}
\label{eq:eps-Kselect}
\varepsilon < \frac{1}{4}\min\bigl(\eta,\; \gout-2\gin,\; 2\gout-\gout^{\max}-\gin\bigr).
\end{equation}
Each of the three terms is strictly positive by
Assumption~\ref{ass:strong-separation}, so such $\varepsilon$ exists. We work on
$\mathcal F_{N,n}(\varepsilon):=\mathcal E_{N,n}(\varepsilon)\cap\{\forall k:C_k\neq\emptyset\}$,
whose complement has probability at most
$N^2\exp(-\varepsilon^2n/(2C^\star))+K(1-w_{\min})^N$ by
Lemma~\ref{lem:uniform} and a union bound. Throughout we work on
$\mathcal F_{N,n}(\varepsilon)$, so that
\begin{equation}
\label{eq:Kselect-bounds}
\hat\gamma_n(i,j)\le\gin+\varepsilon \ \text{ if } Z_i=Z_j,
\qquad
\gout-\varepsilon\le\hat\gamma_n(i,j)\le\gout^{\max}+\varepsilon \ \text{ if } Z_i\neq Z_j.
\end{equation}

At level $t_-:=\gin+\varepsilon$
every pair from the same class is an edge of $G_{t_-}$ and, since
$\gin+\varepsilon<\gout-\varepsilon$ by~\eqref{eq:eps-Kselect}, no
between-class pair is. Hence $G_{t_-}$ has exactly $K$ connected components,
namely the classes $C_1,\ldots,C_K$, all non-empty on
$\mathcal F_{N,n}(\varepsilon)$. Therefore, at least $N-K$ merges occur at height
at most $t_-$, and since the heights are non-decreasing,
\begin{equation}
\label{eq:hNK}
h_{N-K}\;\le\;\gin+\varepsilon .
\end{equation}

Each of the last $K-1$ merges joins two groups $A,B$ that are unions of
distinct classes, and its height is $\min_{i\in A,j\in B}\hat\gamma_n(i,j)$,
a minimum over pairs with $Z_i\neq Z_j$. By~\eqref{eq:Kselect-bounds},
\begin{equation}
\label{eq:htop}
\gout-\varepsilon \;\le\; h_{N-K+1}\;\le\;\cdots\;\le\;h_{N-1}
\;\le\; \gout^{\max}+\varepsilon .
\end{equation}

By~\eqref{eq:hNK}
and~\eqref{eq:htop},
\begin{equation}
\label{eq:transition-gap}
h_{N-K+1}-h_{N-K} \;\ge\; (\gout-\varepsilon)-(\gin+\varepsilon)
\;=\;\gout-\gin-2\varepsilon .
\end{equation}
For $1\le\ell\le N-K-1$ the heights are non-decreasing and non-negative, so
$h_{\ell+1}-h_\ell\le h_{N-K}\le\gin+\varepsilon$. For $N-K+1\le\ell\le N-2$
both heights lie in the interval~\eqref{eq:htop}, so
$h_{\ell+1}-h_\ell\le\gout^{\max}-\gout+2\varepsilon$. Both bounds are
$<\gout-\gin-2\varepsilon$ by~\eqref{eq:eps-Kselect}. The transitional gap is
therefore strictly the largest, and the arg-max in~\eqref{eq:Khat-gap} is
uniquely achieved at $\ell^\star=N-K$, which lies in $\{1,\ldots,N-2\}$ since
$K\ge2$. Hence $\hat K_{N,n}=K$.

By~\eqref{eq:hNK} and~\eqref{eq:htop},
$h_{N-K}\le\gin+\varepsilon<\gout-\varepsilon\le h_{N-K+1}$, so no merge occurs
at a height in $(\gin+\varepsilon,\gout-\varepsilon)$ and
$\hat{\mathcal P}^{\mathrm{SL}}_{N,n}=\mathcal P_{N-K}
=\mathcal P^{\mathrm{SL}}_{N,n}(\gin+\varepsilon)$. The threshold
$t=\gin+\varepsilon$ is admissible in Theorem~\ref{thm:SL}, whose proof shows
that this partition equals $P^\star$ on $\mathcal E_{N,n}(\varepsilon)$.

Both conclusions therefore hold on $\mathcal F_{N,n}(\varepsilon)$, and the
stated bound follows : under Assumption~\ref{ass:growth} both terms tend to $0$.
\end{proof}

\begin{proof}[Proof of Lemma~\ref{lem:kmedoid-one-per-cluster}]
Let $M$ be a minimiser of $\Phi_{N,n}$, $\Phi_i(M):=\min_{m\in M}\hat\gamma_n(i,m)$, and set $u:=\gin+\varepsilon$, $v:=\gout-\varepsilon$. Note that $\varepsilon<\eta/2$ implies $u<v$. On $\mathcal E_{N,n}(\varepsilon)$,
\begin{equation}
\label{eq:kmedoid-bounds}
\hat\gamma_n(i,j)\le u\quad\text{if } Z_i=Z_j,\qquad \hat\gamma_n(i,j)\ge v\quad\text{if } Z_i\neq Z_j.
\end{equation}
On $\mathcal W_N(\delta_0)$ every class is non-empty, and
\begin{equation}
\label{eq:what-bounds}
\hat w_\ell\ge w_{\min}-\delta_0,\qquad \hat w_\ell\le w_{\max}+\delta_0\qquad\text{for all }\ell.
\end{equation}

Suppose for contradiction that some class $C_k$ contains no medoid of $M$. The $K$ medoids then all lie in the $K-1$ classes other than $C_k$. By the
pigeonhole principle, two of them lie in the same class: there exist
$m_1,m_2\in M$ with $Z_{m_1}=Z_{m_2}=:k'\neq k$. Fix any $j\in C_k$ (non-empty on $\mathcal W_N(\delta_0)$) and set $M':=(M\setminus\{m_2\})\cup\{j\}$. We bound $\Phi_i(M)-\Phi_i(M')$ on each class.

\emph{Case $Z_i\notin\{k,k'\}$.} Let $m^\dagger\in\arg\min_{m\in M}\hat\gamma_n(i,m)$.
\begin{itemize}[leftmargin=*,topsep=2pt]
\item If $m^\dagger\neq m_2$, $m^\dagger\in M\cap M'\subseteq M'$, so $\Phi_i(M')\le\hat\gamma_n(i,m^\dagger)=\Phi_i(M)$, hence $\Phi_i(M)-\Phi_i(M')\ge 0$.
\item If $m^\dagger=m_2$, then $m_1\in M'$ with $Z_{m_1}=k'=Z_{m_2}$. By~\eqref{eq:kmedoid-bounds}, both $\hat\gamma_n(i,m_1)$ and $\hat\gamma_n(i,m_2)$ lie in $[\Gamma_{Z_i k'}-\varepsilon,\Gamma_{Z_i k'}+\varepsilon]$, so $\hat\gamma_n(i,m_1)\le\hat\gamma_n(i,m_2)+2\varepsilon$. Therefore $\Phi_i(M')\le\hat\gamma_n(i,m_1)\le\Phi_i(M)+2\varepsilon$, and $\Phi_i(M)-\Phi_i(M')\ge -2\varepsilon$.
\end{itemize}
In either subcase, $\Phi_i(M)-\Phi_i(M')\ge -2\varepsilon$.

\emph{Case $Z_i=k$.} No medoid of $M$ lies in $C_k$, so for every $m\in M$, $Z_m\neq k=Z_i$, hence $\hat\gamma_n(i,m)\ge v$ by~\eqref{eq:kmedoid-bounds}. Thus $\Phi_i(M)\ge v$. On the other hand $j\in M'\cap C_k$ and $Z_i=k=Z_j$, so $\hat\gamma_n(i,j)\le u$, hence $\Phi_i(M')\le u$. Therefore $\Phi_i(M)-\Phi_i(M')\ge v-u$.

\emph{Case $Z_i=k'$.} Since $m_1\in M'$ and $Z_{m_1}=k'=Z_i$, \eqref{eq:kmedoid-bounds} gives $\Phi_i(M')\le\hat\gamma_n(i,m_1)\le u$. As $\Phi_i(M)\ge0$, we get directly
\[
\Phi_i(M)-\Phi_i(M')\;\ge\;-u .
\]


Summing the per-index bounds and dividing by $N$:
\begin{align}
\Phi_{N,n}(M)-\Phi_{N,n}(M')
&= \frac1N\sum_{i=1}^N\bigl(\Phi_i(M)-\Phi_i(M')\bigr)\notag\\
&\ge\; \hat w_k(v-u)-\hat w_{k'}u-2\varepsilon(1-\hat w_k-\hat w_{k'})\notag\\
&\ge\; \hat w_k(v-u)-\hat w_{k'}u-2\varepsilon.
\label{eq:kmedoid-aggregate}
\end{align}

By~\eqref{eq:what-bounds} and $v-u>0$,
\[
\hat w_k(v-u)-\hat w_{k'}u-2\varepsilon
\;\ge\; \bigl[w_{\min}(v-u)-w_{\max}u-2\varepsilon\bigr]-\delta_0 v ,
\]
and, substituting $v-u=\gout-\gin-2\varepsilon$ and $u=\gin+\varepsilon$, the
bracket equals $A-\varepsilon(2w_{\min}+w_{\max}+2)$ with $A$ as
in~\eqref{eq:Adef}. Since $K\ge2$ gives $w_{\min}\le1/2$, we have
$2w_{\min}+w_{\max}+2\le3+w_{\max}$, so the bracket exceeds $A/2$
by~\eqref{eq:eps-kmedoids}. As $v\le\gout\le M$, the correction term satisfies
$\delta_0v\le\delta_0M\le A/4$. Altogether
\[
\Phi_{N,n}(M)-\Phi_{N,n}(M')\;>\;\frac{A}{2}-\frac{A}{4}\;>\;0 ,
\]
contradicting the minimality of $M$. Hence every class contains at least one
medoid; since all $K$ classes are non-empty on $\mathcal W_N(\delta_0)$ and
$|M|=K$, each contains exactly one.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:kmedoids}]
Work on $\mathcal E_{N,n}(\varepsilon)\cap\mathcal W_N(\delta_0)$. By Lemma~\ref{lem:kmedoid-one-per-cluster}, every minimiser $M$ has exactly one medoid in each class. In that case, for any $i\in\{1,\ldots,N\}$:
\begin{itemize}[leftmargin=*,topsep=2pt]
\item the unique medoid $m\in M\cap C_{Z_i}$ satisfies $\hat\gamma_n(i,m)\le u$ by~\eqref{eq:kmedoid-bounds};
\item any other medoid $m'\in M\setminus C_{Z_i}$ satisfies $Z_{m'}\neq Z_i$ and hence $\hat\gamma_n(i,m')\ge v>u$.
\end{itemize}
Consequently the closest medoid of $i$ is the one in $C_{Z_i}$, and the assignment partition $\hat P_{N,n}$ coincides with $P^\star$ (up to relabelling of medoids). Therefore
\[
\PP(\hat P_{N,n}\neq P^\star) \;\le\; \PP(\mathcal E_{N,n}(\varepsilon)^c) + \PP(\mathcal W_N(\delta_0)^c)
\;\le\; N^2\exp\!\Bigl(-\frac{\varepsilon^2n}{2C^\star}\Bigr)+2K\exp\bigl(-2N\delta_0^2\bigr),
\]
by Lemma~\ref{lem:uniform} and~\eqref{eq:Wbound}. Under Assumption~\ref{ass:growth} both terms tend to $0$.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:recovery}]
Substituting $n\ge 4C^\star(1+\theta)\varepsilon^{-2}\log N$ into the bound of Lemma~\ref{lem:uniform},
\[
N^2\exp\!\Bigl(-\frac{\varepsilon^2 n}{2C^\star}\Bigr) \;\le\; N^2\exp\!\bigl(-2(1+\theta)\log N\bigr) \;=\; N^{-2\theta}.
\]
Combined with the condition $n\ge n_0(\varepsilon)$ required by
Lemma~\ref{lem:uniform}, this gives the single-linkage statement via
Theorem~\ref{thm:SL}. For $K$-medoids, Theorem~\ref{thm:kmedoids} adds the term
$2K\exp(-2N\delta_0^2)$.
\end{proof}

\subsection{Proofs for Section~\ref{sec:extensions}}
\begin{proof}[Proof of Proposition~\ref{prop:gen-concentration}]
\emph{(i)} Assume the hidden chains stationary. Then
$(Z^{(k)}_t,\xi_t)_{t\ge1}$ is stationary and mixing, as the product of a
uniformly ergodic finite chain with an i.i.d.\ sequence, so its factor
$X^{(k)}_t=\varphi_k(Z^{(k)}_t,\xi_t)$ is stationary and ergodic; likewise for
$X^{(k',\prime)}$, independent of it by construction. Step~1 of the proof of
Proposition~\ref{prop:convergence} applies verbatim with $M^{\mathrm{mc}}$ in
place of $M$, and gives the almost sure convergence to
$\gamma^{\mathrm{obs}}(k,k')=\inf_n\E[D_n]/n$. For arbitrary initial laws,
Step~2 gives the absolute continuity of the law of the hidden pair with respect
to its stationary counterpart, and the observed pair is a measurable function
of the hidden pair and the noise; the convergence event therefore has full
probability under both, with the same limit.

\emph{(ii)} Write $\widetilde Z_t:=(Z^{(k)}_t,Z^{(k',\prime)}_t)$, a Markov
chain on the finite set $\mathcal Z\times\mathcal Z$. The noise being drawn
independently at each step, the observed pairs are conditionally independent
given $\widetilde Z$: the observed pair process is a hidden Markov chain in the
sense of~\cite[Example~2.16]{paulin}, with underlying chain $\widetilde Z$.
Changing one coordinate of $\widetilde Z_{1:n}$ changes at most one symbol in
each observed sequence, so Step~1 of the proof of
Proposition~\ref{prop:concentration}, applied with $M^{\mathrm{mc}}$, gives
bounded differences $c_i=2M^{\mathrm{mc}}/n$. By~\cite[Corollary~2.17]{paulin},
\eqref{eq:paulin-generic} then holds with $\tmix^{\mathrm{hid}}$ in place of
$\tmix$, and centring as in the proof of
Proposition~\ref{prop:concentration} gives the stated bound.
\end{proof}
\begin{proof}[Proof of Theorem~\ref{thm:gen-clustering}]
As noted at the beginning of Section~\ref{sec:extensions}, the proofs of
Theorems~\ref{thm:SL} and~\ref{thm:kmedoids} use the observation process only
through Lemma~\ref{lem:uniform}, itself a consequence of
Proposition~\ref{prop:concentration} and a union bound; everything else is
deterministic given $\mathcal E_{N,n}(\varepsilon)$ and
$\mathcal W_N(\delta_0)$. Replacing Proposition~\ref{prop:concentration} by
Proposition~\ref{prop:gen-concentration}, and $C^\star$ by $C^{\mathrm{gen}}$,
therefore gives both statements, and likewise the recovery threshold.
\end{proof}
